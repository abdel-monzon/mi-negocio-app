    <script>
    // ============================================================

    // ---------- CONSTANTES ----------
    const PIN_ADMIN = '1234';
    const CORREO_ADMIN = 'monzonabdel@gmail.com';
    let UMBRAL_STOCK = 5;

    // ---------- DATOS ----------
    function getProductos() {
        try { return JSON.parse(localStorage.getItem('inventario_productos')) || []; } catch { return []; }
    }
    function guardarProductos(p) { localStorage.setItem('inventario_productos', JSON.stringify(p)); }

    function getHistorial() {
        try { return JSON.parse(localStorage.getItem('historial_ventas')) || []; } catch { return []; }
    }
    function guardarHistorial(h) { localStorage.setItem('historial_ventas', JSON.stringify(h)); }

    function getCategorias() {
        try { return JSON.parse(localStorage.getItem('categorias')) || []; } catch { return []; }
    }
    function guardarCategorias(c) { localStorage.setItem('categorias', JSON.stringify(c)); }

    function getClientes() {
        try { return JSON.parse(localStorage.getItem('clientes')) || []; } catch { return []; }
    }
    function guardarClientes(c) { localStorage.setItem('clientes', JSON.stringify(c)); }

    function getProveedores() {
        try { return JSON.parse(localStorage.getItem('proveedores')) || []; } catch { return []; }
    }
    function guardarProveedores(p) { localStorage.setItem('proveedores', JSON.stringify(p)); }

    function getGastos() {
        try { return JSON.parse(localStorage.getItem('gastos')) || []; } catch { return []; }
    }
    function guardarGastos(g) { localStorage.setItem('gastos', JSON.stringify(g)); }

    function getMermas() {
        try { return JSON.parse(localStorage.getItem('mermas')) || []; } catch { return []; }
    }
    function guardarMermas(m) { localStorage.setItem('mermas', JSON.stringify(m)); }

    // ---------- SESIÓN ----------
    let rolActual = null;
    let vendedorNombre = '';

    // ---------- LOGIN ----------
    function mostrarLoginAdmin() {
        document.getElementById('loginScreen').classList.add('hidden');
        document.getElementById('loginAdmin').classList.remove('hidden');
        document.getElementById('loginVendedor').classList.add('hidden');
    }
    function mostrarLoginVendedor() {
        document.getElementById('loginScreen').classList.add('hidden');
        document.getElementById('loginVendedor').classList.remove('hidden');
        document.getElementById('loginAdmin').classList.add('hidden');
    }
    function volverLogin() {
        document.getElementById('loginScreen').classList.remove('hidden');
        document.getElementById('loginAdmin').classList.add('hidden');
        document.getElementById('loginVendedor').classList.add('hidden');
    }

    function verificarPIN() {
        const pin = document.getElementById('pinInput').value;
        if (pin === PIN_ADMIN) { entrarAdmin(); } else { mostrarMensaje('❌ PIN incorrecto'); document.getElementById('pinInput').value = ''; }
    }

    function entrarAdmin() {
        rolActual = 'admin';
        document.getElementById('loginAdmin').classList.add('hidden');
        document.getElementById('adminSection').classList.remove('hidden');
        document.getElementById('vendedorSection').classList.add('hidden');
        renderizarAdmin();
        cambiarPestanaAdmin('inventario');
        mostrarMensaje('👑 Bienvenido Administrador');
    }

    function entrarVendedor() {
        const nombre = document.getElementById('vendedorNombre').value.trim();
        if (!nombre) { mostrarMensaje('⚠️ Escribe tu nombre'); return; }
        vendedorNombre = nombre;
        rolActual = 'vendedor';
        document.getElementById('loginVendedor').classList.add('hidden');
        document.getElementById('vendedorSection').classList.remove('hidden');
        document.getElementById('adminSection').classList.add('hidden');
        document.getElementById('nombreVendedorDisplay').textContent = vendedorNombre;
        renderizarVendedor();
        cambiarPestanaVendedor('vender');
        mostrarMensaje('👤 Bienvenido ' + vendedorNombre);
    }

    function cerrarSesion() {
        if (confirm('¿Cerrar sesión?')) {
            rolActual = null;
            document.getElementById('adminSection').classList.add('hidden');
            document.getElementById('vendedorSection').classList.add('hidden');
            document.getElementById('loginScreen').classList.remove('hidden');
            document.getElementById('pinInput').value = '';
            document.getElementById('vendedorNombre').value = '';
            mostrarMensaje('🔒 Sesión cerrada');
        }
    }

    // ---------- ADMIN: INVENTARIO ----------
    function renderizarAdmin() {
        const productos = getProductos();
        const busqueda = document.getElementById('buscarAdmin').value.toLowerCase().trim();
        const filtrados = busqueda === '' ? productos : productos.filter(p => 
            p.nombre.toLowerCase().includes(busqueda) || 
            (p.categoria && p.categoria.toLowerCase().includes(busqueda)) ||
            (p.subcategoria && p.subcategoria.toLowerCase().includes(busqueda))
        );
        const tbody = document.getElementById('cuerpoAdmin');
        if (filtrados.length === 0) {
            tbody.innerHTML = `<tr><td colspan="5" class="text-center" style="padding:20px; color:#999;">${productos.length === 0 ? 'Sin productos' : 'No hay coincidencias'}</td></tr>`;
        } else {
            tbody.innerHTML = filtrados.map((p, index) => {
                const realIndex = productos.indexOf(p);
                const bajoStock = p.stock <= UMBRAL_STOCK;
                return `
                    <tr>
                        <td><strong>${p.nombre}</strong><br><small style="color:#666;">${p.categoria || 'General'}${p.subcategoria ? ' → ' + p.subcategoria : ''}</small></td>
                        <td>$${p.precioCompra.toFixed(2)}</td>
                        <td>$${p.precioVenta.toFixed(2)}</td>
                        <td><span class="badge-stock ${bajoStock ? 'badge-bajo' : ''}">${p.stock} ${bajoStock ? '⚠️' : ''}</span></td>
                        <td>
                            <div class="acciones-botones">
                                <button class="btn btn-primary btn-sm" onclick="abrirModalEditarProducto(${realIndex})">✏️</button>
                                <button class="btn btn-warning btn-sm" onclick="abrirModalMovimientoAdmin(${realIndex})">📦</button>
                                <button class="btn btn-danger btn-sm" onclick="confirmarEliminarAdmin(${realIndex})">🗑️</button>
                                <button class="btn btn-danger btn-sm" onclick="abrirModalMermaRapida(${realIndex})" style="background:#c62828;">📉</button>
                            </div>
                        </td>
                    </tr>
                `;
            }).join('');
        }
        let costoTotal = 0, ventaTotal = 0;
        productos.forEach(p => { costoTotal += p.precioCompra * p.stock; ventaTotal += p.precioVenta * p.stock; });
        document.getElementById('totalCosto').textContent = '$' + costoTotal.toFixed(2);
        document.getElementById('totalVenta').textContent = '$' + ventaTotal.toFixed(2);
        document.getElementById('gananciaPotencial').textContent = '$' + (ventaTotal - costoTotal).toFixed(2);

        // Alertas de stock bajo
        const productosBajos = productos.filter(p => p.stock <= UMBRAL_STOCK);
        const alertaDiv = document.getElementById('alertasStock');
        if (productosBajos.length > 0) {
            alertaDiv.classList.remove('hidden');
            document.getElementById('listaAlertas').textContent = productosBajos.map(p => `${p.nombre} (${p.stock})`).join(', ');
        } else {
            alertaDiv.classList.add('hidden');
        }
        mostrarHistorialAdmin();
    }

    // ---------- ADMIN: AGREGAR PRODUCTO (con categorías y subcategorías) ----------
    function abrirModalAgregar() {
        const categorias = getCategorias();

    function movimientoAdmin(tipo) {
        const cantidad = parseInt(document.getElementById('f_cantidad_admin').value);
        const vendedor = document.getElementById('f_vendedor_admin').value.trim() || 'Admin';
        const clienteData = document.getElementById('f_cliente_admin').value;
        const cliente = clienteData ? { nombre: clienteData.split('|')[0], telefono: clienteData.split('|')[1] } : null;
        if (!cantidad || cantidad <= 0) return mostrarMensaje('⚠️ Ingresa cantidad válida');
        const productos = getProductos();
        const p = productos[indiceAdmin];
        if (tipo === 'salida') {
            if (p.stock < cantidad) return mostrarMensaje('❌ Stock insuficiente');
            p.stock -= cantidad;
            const historial = getHistorial();
            const ahora = new Date();
            const fecha = ahora.toLocaleDateString('es-CU', { day:'2-digit', month:'2-digit', year:'numeric' });
            const hora = ahora.toLocaleTimeString('es-CU', { hour:'2-digit', minute:'2-digit' });
            const venta = {
                producto: p.nombre,
                categoria: p.categoria || 'General',
                subcategoria: p.subcategoria || '',
                cantidad: cantidad,
                precioVenta: p.precioVenta,
                precioCompra: p.precioCompra,
                gananciaUnidad: p.precioVenta - p.precioCompra,
                gananciaTotal: (p.precioVenta - p.precioCompra) * cantidad,
                vendedor: vendedor,
                cliente: cliente ? cliente.nombre : null,
                clienteTelefono: cliente ? cliente.telefono : null,
                fecha: fecha,
                hora: hora,
                timestamp: ahora.getTime()
            };
            historial.push(venta);
            guardarHistorial(historial);
            mostrarTicket(venta);
        } else {
            p.stock += cantidad;
        }
        guardarProductos(productos);
        cerrarModalAdmin();
        renderizarAdmin();
        mostrarMensaje(`✅ ${tipo === 'entrada' ? 'Entrada' : 'Venta'} registrada. Stock: ${p.stock}`);
    }

    function exportarDatos() {
        const data = JSON.stringify({
            productos: getProductos(),
            historial: getHistorial(),
            categorias: getCategorias(),
            clientes: getClientes(),
            proveedores: getProveedores(),
            gastos: getGastos(),
            mermas: getMermas()
        });
        const blob = new Blob([data], {type: 'application/json'});
        const url = URL.createObjectURL(blob);
        const a = document.createElement('a');
        a.href = url;
        a.download = 'respaldo_completo.json';
        a.click();
        URL.revokeObjectURL(url);
        mostrarMensaje('📤 Respaldo completo descargado');
    }

    function importarDatos() {
        const input = document.createElement('input');
        input.type = 'file';
        input.accept = 'application/json';
        input.onchange = function(e) {
            const file = e.target.files[0];
            if (!file) return;
            const reader = new FileReader();
            reader.onload = function(ev) {
                try {
                    const data = JSON.parse(ev.target.result);
                    if (data.productos) guardarProductos(data.productos);
                    if (data.historial) guardarHistorial(data.historial);
                    if (data.categorias) guardarCategorias(data.categorias);
                    if (data.clientes) guardarClientes(data.clientes);
                    if (data.proveedores) guardarProveedores(data.proveedores);
                    if (data.gastos) guardarGastos(data.gastos);
                    if (data.mermas) guardarMermas(data.mermas);
                    renderizarAdmin();
                    mostrarMensaje('📥 Datos restaurados correctamente');
                } catch { mostrarMensaje('⚠️ Error al leer archivo'); }
            };
            reader.readAsText(file);
        };
        input.click();
    }

        const clientes = getClientes();
        if (clientes.length > 0) {
            mensaje += `%0A👥 *CLIENTES:* ${clientes.length} registrados%0A`;
        }
        const proveedores = getProveedores();
        if (proveedores.length > 0) {
            mensaje += '%0A🏢 *PROVEEDORES:*%0A';
            proveedores.forEach(p => {
                mensaje += `• ${p.nombre}${p.telefono ? ' - 📞' + p.telefono : ''}%0A`;
            });
        }
        const gastos = getGastos();
        const totalGastos = gastos.reduce((acc, g) => acc + g.monto, 0);
        if (gastos.length > 0) {
            mensaje += `%0A💸 *GASTOS TOTALES:* $${totalGastos.toFixed(2)}%0A`;
        }
        const totalMermas = mermas.reduce((acc, m) => acc + m.costo, 0);
        if (mermas.length > 0) {
            mensaje += `📉 *MERMAS TOTALES:* $${totalMermas.toFixed(2)}%0A`;
        }
        const url = `https://wa.me/?text=${mensaje}`;
        window.open(url, '_blank');
        mostrarMensaje('📲 Abriendo WhatsApp...');
    }

    function cerrarModalAdmin() { document.getElementById('modalAdmin').classList.remove('active'); }
    function cerrarModalVendedor() { document.getElementById('modalVendedor').classList.remove('active'); }

    function mostrarMensaje(texto) {
        const existente = document.querySelector('.mensaje-flotante');
        if (existente) existente.remove();
        const div = document.createElement('div');
        div.className = 'mensaje-flotante';
        div.textContent = texto;
        document.body.appendChild(div);
        setTimeout(() => { if (div.parentNode) div.remove(); }, 2500);
    }
